select first_name, department, salary
from worker w1
where salary = (
    select distinct salary
    from worker w2
    where w1.department = w2.department
    order by salary desc
    limit 1 offset 1
); 


SELECT department, AVG(salary) AS avg_salary
FROM employees
GROUP BY department
ORDER BY avg_salary ASC
LIMIT 2;



SELECT *
FROM employees
WHERE department = 'Admin'
ORDER BY salary DESC
LIMIT 1 OFFSET 1;
